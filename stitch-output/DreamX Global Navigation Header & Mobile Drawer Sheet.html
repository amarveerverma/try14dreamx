export default function RootLayout({ children }: { children: React.ReactNode }) {
  const [isMobileOpen, setIsMobileOpen] = useState(false);

  return (
    <html lang="en" className="h-full">
      <body className="bg-surface font-sans text-on-surface antialiased min-h-screen">
        {/* Sticky Shell Navigation Header */}
        <header className="sticky top-0 z-40 w-full bg-surface-bright/90 backdrop-blur-md shadow-xs">
          <div className="max-w-7xl mx-auto px-6 h-18 flex items-center justify-between">
            <DreamXLogo />
            <DesktopNav className="hidden lg:flex" />
            <HeaderActionGroup onMobileToggle={() => setIsMobileOpen(true)} />
          </div>
        </header>

        {/* Mobile Accessible Navigation Drawer Sheet */}
        <MobileNavDrawer 
          isOpen={isMobileOpen} 
          onClose={() => setIsMobileOpen(false)} 
          region="INR (₹)" 
        />

        {/* Slot for page-level views */}
        <main className="max-w-7xl mx-auto px-6 py-10">{children}</main>
      </body>
    </html>
  );
}
